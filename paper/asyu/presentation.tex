\section{Part 0 --- the everyday problem}
% ===================================================================
\begin{frame}{The one-sentence problem}
\bigq{A computer reads a receipt and says the total is \textbf{\$58.40}.}
\vspace{1em}
\bigq{\textbf{Should we believe it?}}
\vspace{1em}
\plain{That single question is the whole paper. Everything else is
about answering it well.}
\end{frame}

\begin{frame}{Why it matters}
\begin{itemize}
\item Receipts feed tax filing, expense reports, audits, accounting.
\item A wrong total is not noticed at the time --- it is filed,
trusted, and discovered weeks later, if ever.
\item Billions of receipts are read by computers. Small error rates
become large amounts of money.
\end{itemize}
\vspace{1em}
\keybox{We want an automatic ``\textcolor{bad}{do not trust this
one}'' alarm, so a human checks only the risky receipts.}
\end{frame}

% ===================================================================
\section{Part 1 --- what an ``AI model'' is}
% ===================================================================
\begin{frame}{What is an ``AI model''? (no jargon)}
\plain{A model is just a \textbf{function}: you put something in, you
get something out.}
\vspace{0.8em}
\begin{itemize}
\item In: a picture of a receipt.
\item Out: text fields, including \texttt{total = \$58.40}.
\end{itemize}
\vspace{0.8em}
It was \emph{built} by showing it thousands of example receipts with
the correct answers, and automatically tuning millions of internal
numbers until it usually answers correctly. That tuning is
``training''. We treat the trained model as a fixed black box.
\end{frame}

\begin{frame}{Two model families we use (just names)}
\begin{itemize}
\item \textbf{Donut}: reads the receipt image and \emph{writes the
answer out one character at a time}, like a person typing. Called
``generative''.
\item \textbf{LayoutLMv3}: looks at each word on the page and
\emph{labels} it (``this word is the total''). Called
``encoder-only''.
\end{itemize}
\vspace{1em}
\keybox{We test on both so a finding is not a quirk of one design.}
\end{frame}

\begin{frame}{Probability = a number between 0 and 1}
\plain{A probability is just a fraction of certainty.}
\begin{itemize}
\item $1.0$ = totally sure. \quad $0.5$ = coin flip. \quad $0.0$ = no
chance.
\item $0.97$ means ``pretty sure''. $0.30$ means ``probably not''.
\end{itemize}
\vspace{1em}
That is all ``probability'' will ever mean in this talk: a sureness
dial from 0 to 1.
\end{frame}

\begin{frame}{Softmax / ``confidence'': the model's own sureness}
When the model produces an answer it also produces a number from 0 to
1 about \emph{its own} answer: ``how sure am I?''. The machinery that
makes that number is called \textbf{softmax}; the number is the
model's \textbf{confidence}.
\vspace{1em}
\keybox{\textbf{softmax confidence} = the model grading its own
homework. For 30+ years this self-grade was the only ``should I trust
it?'' signal practitioners had.}
\end{frame}

\begin{frame}{The catch: confidently wrong}
\begin{columns}
\column{0.55\linewidth}
\begin{itemize}
\item Model says \textbf{\$58.40}, ``\textcolor{smx}{97\% sure}''.
\item Real total: \textbf{\$48.40}.
\item It is \emph{sure} and \emph{wrong} at the same time.
\end{itemize}
\column{0.42\linewidth}
\keybox{A self-grade can be confidently wrong. We need a check that
does \emph{not} ask the model about itself.}
\end{columns}
\end{frame}

\begin{frame}{``Distribution shift'': trained here, used there}
\plain{The model learns the patterns of the receipts it was trained
on. Deployed receipts may look different (other shops, countries,
layouts). That mismatch is called \textbf{distribution shift}.}
\vspace{1em}
\begin{itemize}
\item Trained on Indonesian restaurant receipts (a set called
\textbf{CORD}).
\item Used on Malaysian shop receipts (\textbf{SROIE}) or English
receipts (\textbf{WildReceipt}).
\end{itemize}
Under shift, models fail more --- and often \emph{silently}.
\end{frame}

\begin{frame}{The three receipt collections (``datasets'')}
\begin{itemize}
\item \textbf{CORD}: Indonesian restaurant receipts. The ``home''
data.
\item \textbf{SROIE}: Malaysian shop receipts. ``Away'' data.
\item \textbf{WildReceipt}: English receipts. A third ``away'' set,
read by the other model family (LayoutLMv3).
\end{itemize}
\vspace{1em}
\keybox{A ``dataset'' is just a labelled pile of examples: each
receipt plus its true total.}
\end{frame}

\section{Part 2 --- our idea, in one picture}
% ===================================================================
\begin{frame}{The idea: don't ask the AI --- ask the receipt}
\bigq{The receipt already contains every line-item price.}
\vspace{0.6em}
\bigq{If some of them add up to the predicted total, the answer is
arithmetically consistent. If \emph{nothing} adds up, be suspicious.}
\vspace{0.6em}
\plain{We call this arithmetic check \textbf{$\sigma$} (the Greek
letter ``sigma'', traditionally used for ``sum'').}
\end{frame}

\begin{frame}{``Subset-sum'': do some items add to the total?}
Receipt line prices: \;\texttt{3.00, 5.50, 2.00, 48.40, 1.40}.\\
Predicted total: \;\texttt{\$48.40}.
\vspace{0.8em}
\begin{itemize}
\item Question $\sigma$ asks: can I pick \emph{some} of these numbers
that sum exactly to the predicted total (within a cent)?
\item Here: $48.40$ itself works $\Rightarrow$ \textcolor{ok}{consistent}.
\item If predicted were \$58.40 and no subset sums to it
$\Rightarrow$ \textcolor{bad}{flag it}.
\end{itemize}
\keybox{This ``does any subset add up?'' question is the classic
\textbf{subset-sum} problem. A short standard procedure answers it
exactly in microseconds.}
\end{frame}

\begin{frame}{Why $\sigma$ is interesting: it never looks at the AI}
\begin{itemize}
\item softmax: the AI grading itself (inside the model).
\item $\sigma$: arithmetic on the document (outside the model).
\end{itemize}
\vspace{1em}
\keybox{Two \emph{independent} opinions. The whole paper is about
whether combining an inside-opinion and an outside-opinion beats
either alone --- and by how much, honestly measured.}
\end{frame}

\section{Part 3 --- the measuring vocabulary (from zero)}
% ===================================================================
\begin{frame}{We must \emph{measure}. First, four plain words}
\begin{itemize}
\item \textbf{Accept / abstain}: a gate either accepts an answer (``I
trust it'') or abstains (``send to a human'').
\item \textbf{Coverage}: the fraction of receipts the gate accepts.
\item \textbf{Correct}: the accepted answer matched the true total.
\item \textbf{Precision}: among accepted answers, the fraction that
were correct.
\end{itemize}
\vspace{0.8em}
\keybox{High precision but low coverage = ``I only vouch for a few,
but those few I am almost always right about.''}
\end{frame}

\begin{frame}{Average and median (the two ``typical value'' words)}
\begin{itemize}
\item \textbf{Average (mean)}: add the numbers, divide by how many.
\item \textbf{Median}: line the numbers up, take the middle one.
Robust to a few wild values.
\end{itemize}
\vspace{1em}
\keybox{Both answer ``what is a typical value here?''. We use whichever
is less misleading for the quantity at hand.}
\end{frame}

\begin{frame}{Correlation: do two numbers move together?}
\plain{If when one number is big the other tends to be big, they are
\textbf{positively correlated}. If one big means the other small,
\textbf{negatively correlated}. If unrelated, \textbf{near zero}.}
\vspace{0.8em}
A correlation is itself a number from $-1$ to $+1$:
\begin{itemize}
\item $+1$ = move perfectly together, $-1$ = perfectly opposite,
$0$ = unrelated.
\end{itemize}
``Spearman'' correlation is the same idea using \emph{rankings}
(1st, 2nd, 3rd\dots) instead of raw values --- safer when scales
differ.
\end{frame}

\begin{frame}{$\phi$ (phi): correlation for yes/no outcomes}
When the two things are yes/no (e.g.\ ``did $\sigma$ fail?'' and
``did softmax fail?''), the same $-1\ldots+1$ correlation idea has a
special name, \textbf{$\phi$}.
\vspace{0.8em}
\keybox{$\phi \approx 0$ between two error-flags means the two methods
\emph{fail on different receipts} --- exactly the property that makes
combining them useful.}
\end{frame}

\begin{frame}{What does ``significant'' mean? The luck question}
Any measured difference could be a fluke of which receipts we happened
to test. So we always ask:
\bigq{``If there were \emph{no real effect}, how often would pure luck
produce a result this big?''}
That ``how often by luck'' fraction is the \textbf{p-value}.
\begin{itemize}
\item p $=0.05$: luck would do this 1 time in 20. \quad p $=0.001$:
1 in 1000.
\item Small p $\Rightarrow$ unlikely to be luck $\Rightarrow$ we call
it \textbf{statistically significant}.
\end{itemize}
\end{frame}

\begin{frame}{Permutation test: measuring luck by shuffling}
How do we get a p-value without heavy theory? We \emph{simulate luck}:
\begin{enumerate}
\item Randomly shuffle the labels (break any real link on purpose).
\item Re-measure the effect on the shuffled data.
\item Repeat thousands of times $\Rightarrow$ a pile of ``pure luck''
results.
\item p = fraction of luck-results as big as the real one.
\end{enumerate}
\keybox{Honest and assumption-free: we literally show how rare the
real result is compared to chance.}
\end{frame}

\begin{frame}{Bootstrap: how sure are we of a number?}
We measured, say, a lift of $0.05$ on \emph{these} receipts. Would
other receipts give the same? The \textbf{bootstrap}:
\begin{enumerate}
\item Make a new fake sample by drawing receipts at random (with
repeats) from the ones we have.
\item Re-measure. Repeat thousands of times.
\item The spread of those re-measurements shows how shaky the number
is.
\end{enumerate}
\keybox{The middle 95\% of those re-measurements is the
\textbf{95\% confidence interval} --- a plain ``we're fairly sure the
true value is in this range''.}
\end{frame}

\begin{frame}{Reading a confidence interval}
``Lift $=0.05$, 95\% CI $[0.02,\,0.08]$'' means:
\begin{itemize}
\item Best estimate: $0.05$.
\item Plausible range: $0.02$ to $0.08$.
\item The range \emph{excludes 0} $\Rightarrow$ there is probably a
real positive effect.
\item If the range \emph{included 0}, we could not rule out ``nothing''.
\end{itemize}
\keybox{Interval excludes 0 = real; interval straddles 0 = not shown.}
\end{frame}

\begin{frame}{AUROC: one number for ``can this score separate A from B?''}
Give every receipt a risk score. Pick one truly-bad and one truly-good
receipt at random. \textbf{AUROC} = the chance the bad one gets the
higher risk score.
\begin{itemize}
\item $1.0$ = perfect separation. \quad $0.5$ = useless (coin flip).
\item $0.8$ = fairly good. $0.98$ = excellent.
\end{itemize}
\keybox{AUROC turns ``is this score any good at telling apart two
groups?'' into a single 0.5--1.0 grade.}
\end{frame}

\begin{frame}{McNemar: comparing two methods on the \emph{same} items}
Run two methods on the same receipts. Look only at the receipts where
they \emph{disagree}:
\begin{itemize}
\item If disagreements are lopsided (one method right far more often)
$\to$ methods differ.
\item If disagreements are balanced $\to$ ``no asymmetry'': the
methods are not simply better/worse, they are differently-wrong.
\end{itemize}
\keybox{For us, ``no asymmetry'' is good: it means $\sigma$ and
softmax are not the same gate in disguise.}
\end{frame}

\section{Part 4 --- the experiments, in plain words}
% ===================================================================
\begin{frame}{The experiment pipeline (names you will see)}
We run each model on each dataset and record, per receipt:
softmax confidence, whether $\sigma$ accepts, and whether the answer
was actually correct. The scripts are just labels:
\begin{itemize}
\item \textbf{B, MB} = Donut on CORD (home). \quad \textbf{A, M} =
Donut on SROIE (away).
\item \textbf{F, MF2} = LayoutLMv3 on WildReceipt (other model).
\item \textbf{U, U2, U3} = the analyses that compare the gates.
\end{itemize}
\keybox{No model is re-run for the analyses --- U/U2/U3 only
re-read the recorded per-receipt yes/no/correct table.}
\end{frame}

\begin{frame}{Headline measurement: the intersection precision lift}
``Intersection'' = accept only when \emph{both} $\sigma$ and softmax
are happy.
\vspace{0.6em}
\begin{itemize}
\item On \textbf{WildReceipt}: both-agree is right $113$ of $114$ =
\textcolor{ok}{$0.991$}, where each signal \emph{alone} is only
$0.953$.
\item On \textbf{CORD} (test+validation, $n{=}200$, leakage-free):
both-agree $54$ of $55$ = $0.982$, single gates below $0.93$.
\item On \textbf{SROIE}: $15/15$, but $n{=}15$ is too small to claim
more than ``consistent''.
\end{itemize}
\keybox{Where both opinions agree, the answer is almost always right
--- visibly higher than either opinion by itself.}
\end{frame}

\begin{frame}{``But SROIE is only $n{=}15$'' --- the data answers this}
\plain{A reviewer rightly asks: the SROIE intersection cell is tiny.
Does the headline secretly rest on $15$ receipts? We answer with the
same three corpora, the way multi-corpus selective-prediction work
does it --- no new data, no model.}
\vspace{0.5em}
\begin{itemize}
\item \textbf{Honest per-corpus floor (Wilson 95\% lower bound).}
$15/15$ does \emph{not} become $1.000$; it becomes its true floor
\textcolor{sig}{$\pcSroieWilson$}. CORD floor $\pcCordWilson$,
WildReceipt $\pcWrWilson$. The weakest single corpus
(\pcWeakCorpus) is stated straight at $\pcWeakWilson$.
\item \textbf{Corpus-stratified cluster bootstrap} (resample
\emph{within} each corpus, $5000\times$): pooled intersection
precision \textcolor{ok}{$\pcPoolPrec$} over $n{=}\pcPoolN$, 95\% CI
$[\pcBootLo,\pcBootHi]$, Wilson LB $\pcPoolWilson$.
\item \textbf{Leave-one-corpus-out.} Drop SROIE entirely: the claim
still holds on CORD$+$WildReceipt ($169$ intersection accepts, Wilson
LB $0.958$). Dropping \emph{any} corpus leaves the pooled Wilson LB
$\ge\pcLooMinWilson$.
\end{itemize}
\keybox{The intersection claim does not rest on any one corpus ---
throw SROIE away completely and it is unchanged. The small cell is
reported with its honest floor, not hidden; the load-bearing number
is the stratified pooled Wilson LB $\pcPoolWilson$.}
\end{frame}

\begin{frame}{Option 1: exact per-corpus rigor --- stated without spin}
\plain{No SciPy, no PyMC --- exact small-sample statistics from
scratch, on the leakage-free cells (CORD test+validation $54/55$,
SROIE $15/15$, WildReceipt $113/114$).}
\vspace{0.4em}
\begin{itemize}
\item \textbf{Exact lower bounds (Clopper--Pearson / Jeffreys).}
CORD $\rgCordCPlo$/$\rgCordJeff$, SROIE $\rgSroieCPlo$/$\rgSroieJeff$
(the SROIE bound is exactly $0.05^{1/15}$), WildReceipt
$\rgWrCPlo$/$\rgWrJeff$.
\item \textbf{Leave-one-corpus-out.} Worst-case pooled Wilson LB
$\rgLocoMinWil$ (dropping WildReceipt); receipt-weighted pool
$\rgPoolPhat$, Wilson LB $\rgPoolWil$.
\item \textbf{Exact-binomial TOST vs.\ $0.95$ (H$_0$: $p\le0.95$).}
\textcolor{ok}{Passes individually only on WildReceipt}
($p=\rgTostWrP$). \textcolor{bad}{CORD ($p=\rgTostCordP$) and SROIE
($p=\rgTostSroieP$) do not} --- consistent with $>0.95$ but
underpowered. The pool passes ($p=\rgTostPoolP$). We say this
straight.
\item \textbf{Required-$n$ for $0.8$ power:} CORD $\rgReqnCord$,
SROIE $\rgReqnSroie$, WildReceipt $\rgReqnWr$.
\item \textbf{Within-corpus decorrelation (A5, data \emph{was}
present).} Stouffer $p=\rgAFiveStouffer$, Fisher $p=\rgAFiveFisher$:
holds inside every corpus, not just pooled.
\end{itemize}
\keybox{The orthogonality is robust within every corpus. The
$0.95$ ship-without-review floor is met \emph{individually} only
where the sample is large enough to show it --- stated honestly, not
softened.}
\end{frame}

\begin{frame}{The fair objection (and why U/U2 exist)}
\plain{A sceptic says: ``Intersecting \emph{any} two decent filters
raises precision and lowers coverage. Your gain might be that generic
shrink, not real independence.''}
\vspace{1em}
\keybox{So we must prove $\sigma$'s mistakes are genuinely
\emph{decorrelated} from softmax's --- not just ``another filter''.
That is what U and U2 test.}
\end{frame}

\begin{frame}{Two control gates (the honest comparison)}
We compare $\sigma$ against two fake ``placebo'' gates at the same
coverage:
\begin{itemize}
\item \textbf{Random gate}: accepts a random subset. Carries no
information --- isolates the pure ``shrinking coverage'' effect.
\item \textbf{Correlated gate}: a second gate \emph{built from
softmax} with the same standalone quality as $\sigma$. Tests ``would
any second good softmax-like gate do as well?''
\end{itemize}
\keybox{$\sigma$ must beat \emph{both} placebos to claim it adds
genuinely independent information.}
\end{frame}

\begin{frame}{U: the first (too-strict) test}
U asked: does $\sigma$'s confidence interval lie \emph{entirely
above} both placebos' intervals?
\begin{itemize}
\item That non-overlap rule is known to be \emph{over-strict}: two
ranges can overlap while the real difference is still significant.
\item U said ``not cleanly separated''.
\end{itemize}
\keybox{Lesson learned mid-project: U used the wrong test. The fix was
not to soften the claim --- it was to run the \emph{correct} test.}
\end{frame}

\begin{frame}{U2: the correct test (paired difference)}
U2 measures the \emph{difference} directly, receipt-by-receipt, with a
bootstrap, plus a permutation test of error-decorrelation $\phi$.
\vspace{0.6em}
\keybox{Pooled result: $\sigma$'s lift over the random gate
$=\orthDrand$ (p $=\orthDrandP$); over the correlated gate
$=\orthDcorr$ (p $=\orthDcorrP$); error-decorrelation
$\phi=\orthPhi$, permutation p $=\orthPhiP$. Verdict:
\textcolor{ok}{\textbf{\orthEarned}}.}
\vspace{0.4em}
\plain{In words: $\sigma$'s mistakes really are decorrelated from
softmax's --- this is established, not merely suggested.}
\end{frame}

\begin{frame}{What ``\orthEarned'' honestly does and does not mean}
\begin{itemize}
\item \textbf{Does}: the \emph{statistical orthogonality} is real and
reproducible (very small luck-probability, beats both placebos).
\item \textbf{Does not}: it is a \emph{small} effect, and broken out
per dataset it is individually clear only on WildReceipt; on CORD and
SROIE the per-dataset test is not significant.
\end{itemize}
\keybox{We claim the orthogonality. We do \emph{not} claim it is large
or that every dataset shows it alone. Both halves are stated.}
\end{frame}

\begin{frame}{U3: is it operationally \emph{useful}? (honest null)}
``Risk--coverage'': sweep how much you accept; plot how much risk
remains. Compare softmax-alone vs.\ $\sigma$-gated at equal coverage.
\begin{itemize}
\item Pooled: the difference is tiny and its interval \emph{includes
zero} (not significant).
\item Reason: both already sit at \emph{near-zero error} here --- there
is almost no risk left to remove, so the test has no room to show a
gain.
\end{itemize}
\keybox{We keep this null stated. It bounds the \emph{operational}
payoff; it does not weaken the orthogonality claim, which is a
different question.}
\end{frame}

\begin{frame}{Where the operational gain can appear}
A risk reduction can only show up where there is real error to
reduce.
\begin{itemize}
\item WildReceipt --- the one regime with appreciable base error --- is
exactly where $\sigma$-gating is individually favourable.
\item Where the model is already almost always right (CORD home
turf), no gate can improve much.
\end{itemize}
\keybox{Honest scope: orthogonality established; operational benefit
expected (and seen) only when base error is non-trivial.}
\end{frame}

\section{Part 5 --- honest summary}

\begin{frame}{What this paper claims --- exactly}
\begin{itemize}
\item[\textcolor{ok}{$\checkmark$}] A document-side arithmetic check
$\sigma$ is \emph{statistically softmax-orthogonal}: its errors are
decorrelated from the model's confidence (U2: p $=\orthPhiP$, beats
two placebos).
\item[\textcolor{ok}{$\checkmark$}] Where both agree, precision is
visibly higher (e.g.\ WildReceipt $0.991$ vs $0.953$).
\item[\textcolor{ok}{$\checkmark$}] Works across two model families
and three datasets; McNemar shows no asymmetry.
\end{itemize}
\end{frame}

\begin{frame}{What this paper does \emph{not} claim (kept stated)}
\begin{itemize}
\item[\textcolor{bad}{$\times$}] Not a large effect --- the pooled
lift is small.
\item[\textcolor{bad}{$\times$}] Not significant per-dataset on CORD
or SROIE (only WildReceipt individually).
\item[\textcolor{bad}{$\times$}] No proven risk--coverage gain pooled
(near-zero base error leaves no headroom).
\end{itemize}
\keybox{Stating the limits \emph{is} the contribution's credibility.
We neither inflate nor disown them.}
\end{frame}

\begin{frame}{One-slide takeaway}
\bigq{The model's self-confidence is one opinion. Arithmetic on the
receipt is a genuinely \emph{independent} second opinion.}
\vspace{0.8em}
\bigq{Their disagreements do not line up (proven, small p) --- so
combining them is principled, and where it matters most (real error),
it visibly helps.}
\vspace{0.8em}
\plain{Honest, reproducible, and exactly scoped --- no more, no less.}
\end{frame}
