% ==========================================================
% Slide 4: OLD WAY
\begin{frame}{}
\vfill
\begin{center}
{\Large \textbf{The usual way: ask the model how confident it is.}}
\end{center}
\vspace{1.5em}
\begin{center}
\begin{tikzpicture}[scale=1.2]
\node[draw,rounded corners,fill=sig!20,minimum width=3cm,minimum height=1.2cm] (m) at (0,0) {\textbf{model}};
\draw[->,thick] (m.east) -- (3,0);
\node[anchor=west] at (3,0.4) {\Large ``\$58.40''};
\node[anchor=west] at (3,-0.4) {\Large 97\% sure};
\end{tikzpicture}
\end{center}
\vspace{1em}
\begin{center}
{\large ... but what if the real total is \$48.40, \emph{not} \$58.40?}\\[0.5em]
{\Large \textbf{\textcolor{bad}{The model can be confidently wrong.}}}
\end{center}
\vfill
\end{frame}

% ==========================================================
% Slide 5: NEW IDEA
\begin{frame}{}
\vfill
\begin{center}
{\Huge \textbf{The new idea}}\\[1.5em]
{\Large Don't ask the model.}\\[0.5em]
{\Large \textbf{Check the math on the receipt itself.}}
\end{center}
\vspace{1.5em}
\begin{center}
\large The receipt has all the numbers.\\
\large Do any of them \emph{add up} to the prediction?
\end{center}
\vfill
\end{frame}

% ==========================================================
% Slide 6: HOW SIGMA WORKS
\begin{frame}{}
\begin{center}
{\large \textbf{The $\sigma$ gate: find a subset that sums to the predicted total}}
\end{center}
\vspace{0.5em}
\begin{center}
\begin{tikzpicture}[scale=0.85]
\draw[fill=white,thick] (0,0) rectangle (4,5);
\draw (0.3,4.5) -- (3.7,4.5);
\node at (2,4.7) {\small\bfseries RECEIPT};
\node[anchor=west,text=ok] at (0.3,3.9) {\small Coffee \hfill \$3.20 \checkmark};
\node[anchor=west,text=ok] at (0.3,3.4) {\small Sandwich \hfill \$8.50 \checkmark};
\node[anchor=west,text=ok] at (0.3,2.9) {\small Salad \hfill \$6.00 \checkmark};
\node[anchor=west,text=ok] at (0.3,2.4) {\small Pasta \hfill \$30.70 \checkmark};
\draw (0.3,1.9) -- (3.7,1.9);
\node[anchor=west,text=ok] at (0.3,1.3) {\small Tax \hfill \$10.00 \checkmark};
\draw[thick] (0.3,0.8) -- (3.7,0.8);
\node[anchor=west] at (0.3,0.3) {\small\bfseries TOTAL \hfill \$58.40};

\node at (9,3.8) [text=sig,font=\Large] {$3.20 + 8.50 + 6.00$};
\node at (9,3.1) [text=sig,font=\Large] {$+ 30.70 + 10.00$};

\draw[->,thick] (9,2.5) -- (9,1.9);
\node at (9,1.3) [text=ok,font=\huge] {= \$58.40 \checkmark};
\node at (9,0.3) [text=ok,font=\Large] {\textbf{accept}};
\end{tikzpicture}
\end{center}
\vspace{0.5em}
\begin{center}
\small If \emph{no} subset sums to \$58.40 within tolerance, the gate rejects.
\end{center}
\end{frame}
