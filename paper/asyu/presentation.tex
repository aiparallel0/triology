\begin{frame}{Why care about receipt KIE precision?}
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\begin{itemize}
\setlength{\itemsep}{0.6em}
\item KIE pipelines route extracted \texttt{total} directly into bookkeeping, expense, and audit systems.
\item A silently wrong total $=$ a real downstream error.
\item Selective prediction: ``defer the receipts I cannot reliably handle.''
\item The bottleneck is \emph{precision on the accepted slice}, not raw F1.
\end{itemize}

\begin{frame}{The standard signal: softmax confidence}
\begin{block}{Definition}
For a generative KIE model (e.g.\ Donut), softmax confidence is the
geometric mean of per-token probabilities of the generated
\texttt{total} string.
\end{block}

\bigskip
\textbf{Problem:} softmax is a property of the \emph{generator}.

\bigskip
\begin{itemize}
\item When the generator hallucinates a \emph{plausible} total, softmax often remains high.
\item The signal cannot disagree with itself.
\end{itemize}

\bigskip
\textbf{Idea:} introduce a second gate that fails on \emph{different} receipts. Intersect the two.

\bigskip
If their errors are disjoint, intersection precision $\to 1.0$.
\end{frame}

% ------------------------------------------------------------------------
\begin{frame}{$\sigma$: a generator-orthogonal precision gate}
\textbf{Read the receipt, not the logits.}

\bigskip
Given predicted total $\pi$ and multiset $M = \{m_1, \dots, m_n\}$ of money amounts:
\[
\sigma(\pi) = 1 \iff \exists\, S \subseteq M,\ |S| \ge k_{\min},\ 
\left|\textstyle\sum_{i \in S} m_i - \pi\right| \le \varepsilon.
\]

\bigskip
\textbf{Why it works.} Well-formed receipts are \emph{arithmetically closed} ---
the total is a sum of visible amounts.

\bigskip
$T(M)$ = reachable set of subset sums.
$\sigma$ accepts iff some subset reconstructs the model's answer from
the receipt's own contents.
\end{frame}

% ------------------------------------------------------------------------
\begin{frame}{Implementation}
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\textbf{Verifier:} $0/1$-knapsack dynamic program.
\begin{itemize}
\item State $(i, s)$: ``is $s$ reachable using a subset of the first $i$ items?''
\item Integer cents, no floats.
\item Cardinality guard $k_{\min}{=}2$ excludes singleton witnesses
on labeled corpora.
\item Multi-candidate $\tau$ for OCR (mean $3.12$ candidates/receipt on SROIE).
\end{itemize}

\column{0.40\textwidth}
\textbf{Cost.}\\
$p_{99} = 547\,\mu\text{s}$ per receipt (SROIE).\\
Paired vs Donut generate ($175$ ms median):
\[
\text{overhead} = \mathbf{0.0016\%}.
\]
The verifier is \emph{operationally free}.
\end{columns}
\end{frame}

\begin{frame}{F3: regime mechanism}
Single-gate $\sigma$ precision differs across regimes:

\begin{center}
\begin{tabular}{lccc}
\toprule
Corpus & $\sigma$ & softmax & intersect \\
\midrule
CORD       & \textbf{0.982} & 0.927 & 0.970 \\
SROIE      & 0.867 & \textbf{0.947} & 1.000 \\
WildReceipt & 0.953 & 0.953 & \alert{0.991} \\
\bottomrule
\end{tabular}
\end{center}

\bigskip
\textbf{Mechanism.} The regime gap is in $\tau$-extraction quality, not in the verifier:
\begin{itemize}
\item CORD failures: single-money receipts where $|S| \ge 2$ witness is impossible.
\item SROIE failures: OCR-introduced phantom amounts.
\item WildReceipt: $\sigma$ and softmax \emph{equally precise alone}, intersection still wins.
\end{itemize}

\bigskip
\textbf{Takeaway.} $\sigma$'s value is not ``beats softmax'' but ``fails on different inputs.''
\end{frame}

% ------------------------------------------------------------------------
\begin{frame}{Operational recipe \& limitations}
\textbf{Recipe.}
\begin{enumerate}
\item Pick a KIE model; produce $\pi$ and softmax $s$ per receipt.
\item Extract $M$ from the document.
\item Run $\sigma$ at $\varepsilon{=}0.02$. $k_{\min}{=}2$ labeled, $k_{\min}{=}1$ OCR.
\item Accept iff $\sigma = 1$ \emph{and} $s \ge \tau_s$. Defer otherwise.
\end{enumerate}

\bigskip
\textbf{Limitations.}
\begin{itemize}
\item Arithmetic-closed fields only (totals/subtotals/taxes).
\item Single field, two backbones, three corpora --- multi-field $\sigma$ is journal-version.
\item Cardinality-guard inversion measured on SROIE; CORD-side is structural.
\item $\varepsilon$ fixed at $0.02$ (integer-cent regime).
\end{itemize}
\end{frame}

% ------------------------------------------------------------------------
\begin{frame}{Conclusion}
\textbf{Three findings on $n = 919$, two backbones, three regimes.}

\bigskip
\begin{itemize}
\setlength{\itemsep}{0.6em}
\item \textbf{(F1)} Pooled intersection precision Wilson lower bound $\mathbf{0.956}$ ($160/162$); exceeds both single-gate lower bounds.
\item \textbf{(F2)} Paired McNemar non-significance on all three corpora --- the orthogonality the intersection exploits.
\item \textbf{(F3)} Regime gap is in $\tau$-extraction, not the verifier. WildReceipt: equal single-gate precision, intersection still rises.
\end{itemize}

\bigskip
\textbf{Cost.} $0.0016\%$ of Donut generate latency. Operationally free.
