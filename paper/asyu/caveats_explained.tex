\begin{frame}{How to read this deck}
\begin{itemize}
\item Assumes \textbf{only basic arithmetic}: bigger/smaller, fractions,
averages.
\item Every term is defined before it is used.
\item One idea per slide. Every number shown is the real measured value
from the paper, including the unflattering ones.
\item Goal: understand \emph{why} each caveat in the paper cannot be
``fixed'' by editing, because each is a fact about the data, not a typo.
\end{itemize}
\vspace{1em}
\keybox{A caveat is a sentence that says ``here is exactly how far our
result does and does not reach''. Stating it is honesty, not weakness.}
\end{frame}

\section{Part 0: two kinds of problem}
% =====================================================================
\begin{frame}{Two completely different kinds of problem}
\begin{itemize}
\item \textbf{Copy-edit defect}: a number typed wrong, a stale figure, a
contradictory sentence. \textcolor{ok}{Fixable by editing.} We fixed all
of these (including a stale latency figure: the real numbers are later
in this deck).
\item \textbf{Empirical truth}: what the data actually shows, including
where it is weak. \textcolor{bad}{Not fixable by editing}, only by
collecting more/different evidence, or by lying.
\end{itemize}
\vspace{1em}
\keybox{The caveats in this deck are the second kind. You change them
only by doing new science, or by being dishonest. We do neither: we
state them plainly, in the paper itself.}
\end{frame}

\begin{frame}{The paper in one line}
\bigq{$\sigma$ is an arithmetic check: do some of the receipt's printed
line amounts add up to the predicted total (within a cent)?}
\vspace{0.6em}
\plain{$\sigma$ never reads the model. Combined with the model's own
softmax confidence (accept only when \emph{both} agree), precision rises.
This deck is about exactly where that story is strong and where it is
not.}
\end{frame}

% =====================================================================
\section{Part 1: the words you need}
% =====================================================================
\begin{frame}{Precision and coverage (plain)}
\plain{Think of a gate that says ``accept'' or ``flag for a human''.}
\begin{itemize}
\item \textbf{Coverage}: of all receipts, what fraction does the gate
\emph{accept} (not send to a human). Accept 18 of 100 = coverage 0.18.
\item \textbf{Precision}: of the accepted ones, what fraction are
\emph{actually correct}. $\poolIntC$ correct of $\poolIntA$ accepted
$\approx 0.989$.
\end{itemize}
\keybox{High precision on accepted items is the whole point: the gate
should only wave through receipts it is almost never wrong about.}
\end{frame}

\begin{frame}{Sample size $n$ and why it rules everything}
\plain{$n$ is just how many examples you measured.}
\begin{itemize}
\item 15 out of 15 correct sounds perfect. But with only $n{=}15$ you
cannot tell ``truly excellent'' from ``lucky''.
\item More $n$ = a sharper estimate. Tiny $n$ = a blurry one, even at
100\%.
\end{itemize}
\keybox{Small $n$ does not make a result wrong. It makes it
\emph{uninformative}: consistent with both ``great'' and ``mediocre''.}
\end{frame}

\begin{frame}{Confidence interval and ``lower bound''}
\plain{A confidence interval is the honest range the true value could be
in. We always quote the pessimistic floor.}
\begin{itemize}
\item 15/15 at $n{=}15$: best guess 1.000, but the 95\% \emph{lower
bound} is only $\rgSroieWil$ (Wilson). The truth could be near 0.80.
\item 113/114 at $n{=}472$: lower bound $\rgWrWil$. Much tighter,
because $n$ is larger.
\end{itemize}
\keybox{We quote the \emph{lower bound}, the pessimistic honest floor,
never the flattering point estimate.}
\end{frame}

\begin{frame}{Statistical power and ``underpowered''}
\plain{Power = the chance your test can \emph{detect} a real effect that
is truly there.}
\begin{itemize}
\item If $n$ is too small, even a real effect fails the test: not
because it is absent, but because the test had no resolution.
\item ``\textbf{Underpowered}'' = ``$n$ too small to conclude'', NOT
``the method failed''. Two very different sentences.
\end{itemize}
\keybox{An underpowered cell is honest evidence of \emph{not knowing
yet}, not evidence \emph{against}.}
\end{frame}

\begin{frame}{TOST: a fair pass/fail at a threshold}
\plain{TOST asks one crisp question: is precision provably above a bar
(here 0.95, the ``ship without a human'' floor)?}
\begin{itemize}
\item It returns a $p$-value. Small $p$ ($<0.05$) = ``yes, provably
above 0.95''. Large $p$ = ``cannot show it at this $n$''.
\item It is deliberately strict: it refuses to pass on small samples.
\end{itemize}
\keybox{TOST is the referee. We report exactly which corpora it passes
and which it does not. No spin.}
\end{frame}

\begin{frame}{Orthogonal, McNemar, and ``the placebo test''}
\begin{itemize}
\item \textbf{Orthogonal}: $\sigma$'s mistakes and softmax's mistakes
fall on \emph{different} receipts, so combining them helps.
\item \textbf{McNemar}: looks only at receipts where the two gates
disagree; ``no asymmetry'' means neither is simply the other in
disguise.
\item \textbf{Placebo gates}: a random gate and a softmax-built gate of
equal quality. $\sigma$ must beat \emph{both} to prove its information
is genuinely independent, not just generic ``filter shrink''.
\end{itemize}
\keybox{These let us separate ``a real independent signal'' from ``any
two filters intersected always look better''.}
\end{frame}

\section{Part 2: Caveat 1: per-corpus underpowered}
% =====================================================================
\begin{frame}{The real headline numbers (intersection)}
\begin{itemize}
\item \textbf{Pooled} ($n{=}\poolN$): $\poolIntC/\poolIntA = 0.989$,
95\% Wilson lower bound \textbf{$\rgPoolWil$}, Clopper--Pearson lower
bound $\rgPoolCPlo$.
\item \textbf{$\sigma$ alone, pooled}: accepts $\poolSigA$, correct
$\poolSigC$ ($\approx 0.946$).
\item \textbf{WildReceipt} ($n{=}\wrN$): $\wrIntC/\wrIntA = 0.991$.
\item \textbf{CORD} ($n{=}200$, test+validation, leakage-free):
$54/55 = 0.982$.
\item \textbf{SROIE} ($n{=}347$): $15/15$, but only \textbf{15}
accepted.
\end{itemize}
\keybox{Looks strong. Now look at \emph{where the evidence lives}.}
\end{frame}

\begin{frame}{The concentration problem (basic)}
\plain{Of the $\poolIntA$ pooled intersection accepts, \textbf{$\wrIntA$}
come from WildReceipt alone.}
\vspace{0.6em}
\begin{itemize}
\item That is about \textbf{62\%} of the headline mass in one corpus.
\item SROIE contributes only \textbf{15} of $\poolIntA$ ($\approx 8\%$).
\end{itemize}
\keybox{The pooled 0.989 is real, but it is \emph{carried by
WildReceipt}. The paper says this and leads with the per-corpus read,
not the pool.}
\end{frame}

\begin{frame}{The small-$n$ problem (intermediate)}
SROIE is $15/15$. Its honest floors at $n{=}15$:
\begin{itemize}
\item Wilson 95\% lower bound: \textbf{$\rgSroieWil$}
\item Clopper--Pearson exact: \textbf{$\rgSroieCPlo$} (exactly
$0.05^{1/15}$)
\item Jeffreys: \textbf{$\rgSroieJeff$}
\end{itemize}
\vspace{0.4em}
\keybox{Even a \emph{perfect} 15/15 only guarantees ``about 0.80 or
better''. That is a \emph{consistency} statement, not strong evidence.}
\end{frame}

\begin{frame}{The referee's verdict (advanced): per-corpus TOST vs 0.95}
One-sided exact-binomial TOST, threshold 0.95, $\alpha{=}0.05$:
\begin{itemize}
\item \textbf{WildReceipt}: $p=\textbf{\rgTostWrP}$ \quad
\textcolor{ok}{passes}
\item \textbf{CORD}: $p=\rgTostCordP$ \quad \textcolor{bad}{does not
pass}
\item \textbf{SROIE}: $p=\rgTostSroieP$ \quad \textcolor{bad}{does not
pass}
\item \textbf{Pool}: $p=\rgTostPoolP$ \quad \textcolor{ok}{passes}
\end{itemize}
\vspace{0.2em}
Leave-one-corpus-out worst Wilson lower bound: \textbf{$\rgLocoMinWil$}.
Within-corpus error-decorrelation still holds everywhere (Stouffer
$p=\rgAFiveStouffer$, Fisher $p=\rgAFiveFisher$).
\keybox{Honest reading: per-corpus the floor is individually proven only
on WildReceipt; CORD and SROIE are \emph{consistent but underpowered}.
Stated straight, not softened.}
\end{frame}

\begin{frame}{Why editing cannot fix Caveat 1}
\plain{To make CORD/SROIE individually pass, you do not need better
words. You need more accepted receipts.}
\begin{itemize}
\item Required $n$ for 80\% power: CORD $\approx$\textbf{\rgReqnCord},
SROIE $\approx$\textbf{\rgReqnSroie}, WildReceipt already there
($\approx$\rgReqnWr).
\item That means: add corpora with enough intersection mass \emph{and}
re-run the models on a GPU. A different, larger paper.
\item Even done, it raises \emph{power}; it does not guarantee a
reviewer \emph{values} a small effect.
\end{itemize}
\keybox{So the paper does the only honest thing: reports the floors, the
leave-one-out, and the exact TOST pass/fail, in plain language.}
\end{frame}

\section{Part 3: Caveat 2: $\sigma$ alone trails softmax in OCR}
% =====================================================================
\begin{frame}{The honest single-gate picture}
\plain{As a \emph{single} gate (not intersected), $\sigma$'s strength
tracks how clean the printed amounts are.}
\begin{itemize}
\item \textbf{CORD} (clean labelled amounts): $\sigma$ \emph{beats}
softmax, \textbf{0.990} vs \textbf{0.960} at matched coverage.
\item \textbf{WildReceipt} (per-token labels): the two \emph{tie} at
\textbf{0.953}.
\item \textbf{SROIE} (OCR-derived amounts, the regime most like
production): $\sigma$ \textcolor{bad}{\emph{trails} softmax,
\textbf{0.863} vs \textbf{0.945}}.
\end{itemize}
\keybox{We say this plainly: in the messy OCR regime $\sigma$ \emph{alone}
is worse than softmax alone. Its value there is purely in the
intersection.}
\end{frame}

\begin{frame}{Why this caveat is honest, not a bug}
\plain{The SROIE loss is consistent with upstream OCR, not the verifier:
digit confusions and phantom amounts inflate the reachable-sum set with
coincidental near-misses.}
\begin{itemize}
\item This is stated as a \emph{hypothesis} in the paper, not a proven
mechanism.
\item The clean controlled test (hand-OCR-corrected SROIE subsample)
is explicitly named as \emph{future work}.
\end{itemize}
\keybox{We do not claim $\sigma$ is a stronger single gate everywhere.
The honest claim is: it helps most exactly where clean amounts are
available, and in the intersection even on SROIE.}
\end{frame}

\begin{frame}{Why editing cannot fix Caveat 2}
\begin{itemize}
\item To make $\sigma$-alone beat softmax on SROIE, the upstream OCR
amounts would have to be cleaner. That needs new extraction work and a
re-run, not a wording change.
\item The only way to ``remove'' this caveat by editing is to
\emph{claim} $\sigma$ wins where it loses. That is exactly the
dishonesty this effort exists to prevent.
\item So the paper states the $0.863$ vs $0.945$ straight, and locates
$\sigma$'s value in the intersection and on clean amounts.
\end{itemize}
\keybox{A true sentence that limits your claim is not a defect. A false
sentence that inflates it is.}
\end{frame}

\section{Part 4: Caveat 3: the effect is small, not large}
% =====================================================================
\begin{frame}{The intersection lift is real but \emph{small}}
\plain{We proved $\sigma$'s mistakes are genuinely decorrelated from
softmax's, beating \emph{both} placebo gates.}
\begin{itemize}
\item Pooled lift over the random gate: $\orthDrand$
($p=\orthDrandP$); over the softmax-built gate: $\orthDcorr$
($p=\orthDcorrP$); error-decorrelation $\phi=\orthPhi$ (permutation
$p=\orthPhiP$). Verdict: \textcolor{ok}{\textbf{\orthEarned}}.
\item McNemar is non-significant on every corpus (pooled $b=\poolMcB$,
$c=\poolMcC$, $p=\poolMcP$): no asymmetry, exactly as expected.
\end{itemize}
\keybox{The orthogonality is established, \emph{not} merely suggested.
But the size of the pooled lift is small ($\orthDrand$/$\orthDcorr$),
and we say so.}
\end{frame}

\begin{frame}{The operational gain is modest and costs coverage}
``Risk--coverage'': accept fewer, see how much risk remains. Compare
softmax-alone vs $\sigma$-gated at equal coverage.
\begin{itemize}
\item Pooled AURC $\rcAURCsoft \to \rcAURCsig$; difference
$\rcAURCd$, 95\% CI $[\rcAURCdLo,\rcAURCdHi]$ \emph{excludes zero}
(one-sided $p=\rcAURCp$): a real but \emph{small} gain.
\item The intersection fires on only 18.1\% of receipts vs 38.1\% for
either single gate (SROIE just 4.3\%). The lift is bought with coverage.
\end{itemize}
\keybox{We report a modest, statistically resolved operational gain,
concentrated where base error is non-trivial (WildReceipt). Not a large
win, and we never call it one.}
\end{frame}

\begin{frame}{Why editing cannot fix Caveat 3}
\begin{itemize}
\item ``Make the effect large'' would require the pooled lift to
\emph{actually be} large. It is not; the data says $\orthDrand$.
\item ``Drop the coverage cost'' would require the intersection to fire
more often while staying right. That is a different result, not an edit.
\item Deleting the ``small effect'' sentence without new evidence would
be a false claim.
\end{itemize}
\keybox{We claim the orthogonality. We do \emph{not} claim it is large
or per-corpus universal. Both halves are written into the paper.}
\end{frame}

\section{Part 5: Caveat 4: latency provenance}
% =====================================================================
\begin{frame}{What the latency numbers actually are}
\plain{The arithmetic check is a tiny table computation. The corrected,
real measured numbers:}
\begin{itemize}
\item Median \textbf{\latMedian~\latUnit{}} per receipt; $p_{99}$
\textbf{\latPNineNine~\latUnit{}}; max \textbf{\latMax~\latUnit{}}, over
\textbf{\latN{}} receipts.
\item Microsecond scale: millionths of a second, against \emph{hundreds
of milliseconds} for the neural model on a GPU.
\end{itemize}
\keybox{Earlier drafts carried different (stale) figures. These are the
single-sourced, corrected values from the run artifact.}
\end{frame}

\begin{frame}{The latency caveat, stated explicitly}
\plain{The honesty here is in the \emph{provenance}, not just the
numbers.}
\begin{itemize}
\item Timed \textbf{standalone on a CPU} (\latHardware), running the
exact headline DP ($\varepsilon{=}0.02$, the $k_{\min}$ rule).
\item Timed at the \emph{real} per-receipt amount counts for
SROIE${+}$WildReceipt only. CORD's per-receipt counts were not
persisted, so CORD is \textbf{excluded, not fabricated}.
\item We did \textbf{not} measure the end-to-end GPU pipeline on this
host, so we claim \textbf{no ``\% of inference''} figure: only the
microsecond scale and the qualitative ``negligible''.
\end{itemize}
\keybox{This is a faithful compute reconstruction at the real input
sizes, not a replay of stored values and not an end-to-end ratio. We
say exactly that.}
\end{frame}

\begin{frame}{Why editing cannot fix Caveat 4}
\begin{itemize}
\item To claim ``$\sigma$ is X\% of inference'' you must measure the
real neural pipeline and $\sigma$ on the \emph{same} hardware. This host
is CPU-only; that measurement does not exist.
\item Writing a percentage anyway would be fabrication, the exact thing
this effort removes.
\item So the paper reports the standalone microsecond scale with full
provenance, and stops there.
\end{itemize}
\keybox{``Negligible'' is justified by the microsecond scale. A specific
percentage is not justified, so we do not state one.}
\end{frame}

\section{Part 6: Caveat 5: scope and novelty}
% =====================================================================
\begin{frame}{One field, one operating point, one configuration}
\begin{itemize}
\item One field (the total), two backbones (Donut, LayoutLMv3), three
corpora. Multi-field $\sigma$ is journal-scale future work.
\item One operating point: $\varepsilon{=}0.02$, the label-free
$k_{\min}$ rule. The $k_{\min}{=}2$/$k_{\min}{=}1$ ablation is the only
departure and is explicitly an ablation.
\item $\sigma$ assumes the prediction is a parsed numeric scalar;
format-hallucinated outputs need an upstream normaliser the paper does
\emph{not} benchmark.
\end{itemize}
\keybox{$k_{\min}$ is \textbf{not} a flat global 2: it is $k_{\min}{=}2$
on CORD and WildReceipt, $k_{\min}{=}1$ on SROIE, chosen without labels
from how often a single amount equals the prediction.}
\end{frame}

\begin{frame}{Novelty is deliberately scoped}
\plain{Arithmetic cross-checks of receipt totals are long-standing
engineering practice. We do not claim we invented the idea of checking
arithmetic.}
\begin{itemize}
\item The contribution is the \emph{peer-reviewed formalisation} of such
a check as a softmax-orthogonal selective-prediction signal, with paired
and control statistics across two architectures.
\item This is an \textbf{applied} contribution. We claim \emph{no}
methodological novelty in the verifier algorithm itself (it is a
textbook 0/1-knapsack DP).
\end{itemize}
\keybox{Scoping the novelty honestly is part of the credibility, not a
weakness to be hidden.}
\end{frame}

\section{Part 7: the meta-point}
% =====================================================================
\begin{frame}{What we fixed vs what we kept}
\begin{itemize}
\item \textcolor{ok}{Fixed (copy-edit defects):} a leakage-prone
evaluation (CORD now test+validation only), a stale latency figure,
internally contradictory wording, em-dashes, column overflows.
\item \textcolor{bad}{Kept and stated (empirical truths):} per-corpus
underpowered (CORD/SROIE); $\sigma$ alone trails softmax in the OCR
regime ($0.863$ vs $0.945$); the effect is small and costs coverage;
latency provenance is CPU-only with no end-to-end ratio; single
configuration; an applied formalisation, no methodological-novelty
claim.
\end{itemize}
\keybox{The first list is now empty. The second list is, deliberately,
written into the paper in plain language.}
\end{frame}

\begin{frame}{Why this is the strong position, not the weak one}
\begin{itemize}
\item A reviewer who finds a hidden weakness rejects you and distrusts
the rest. A reviewer who sees you state it first trusts the rest.
\item Every number here is the real measured value, including the
unflattering ones (0.863, the small lift, the CPU-only latency).
\item No paper is a guaranteed accept. Stating limits precisely is the
most defensible posture for an applied venue.
\end{itemize}
\keybox{Honest, reproducible, and exactly scoped: no more, and no less,
than the data supports.}
\end{frame}

\begin{frame}{One-slide summary of every caveat}
\begin{itemize}
\item \textbf{C1 underpowered}: pooled $\poolIntC/\poolIntA$ carried by
WildReceipt; per-corpus TOST passes only WildReceipt
($\rgTostWrP$ / $\rgTostCordP$ / $\rgTostSroieP$). Needs data + GPU.
\item \textbf{C2 OCR regime}: $\sigma$ alone trails softmax on SROIE
($0.863$ vs $0.945$). Hypothesised upstream OCR; controlled test is
future work.
\item \textbf{C3 small effect}: pooled lift $\orthDrand$/$\orthDcorr$,
modest AURC gain, costs coverage. Orthogonality real, not large.
\item \textbf{C4 latency provenance}: median \latMedian~\latUnit{},
$p_{99}$ \latPNineNine~\latUnit{}, max \latMax~\latUnit{}, standalone
CPU at real input sizes; no \% of inference claimed.
\item \textbf{C5 scope}: one field, one configuration, applied
formalisation, no methodological-novelty claim.
\end{itemize}
\keybox{They are facts. We state them. That is the contribution's
credibility.}
\end{frame}
